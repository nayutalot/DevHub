## 5. 凭据加密（credentials.json 值的 enc:v1 信封）
scheme  = "enc:v1:" + b64url(iv12) + "." + b64url(authTag16) + "." + b64url(ciphertext)   // 注意顺序: IV.TAG.CT
algo    = aes-256-gcm
key     = sha256(credentialSecret)
secret  = env.ZCODE_CREDENTIAL_SECRET
